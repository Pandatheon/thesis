\chapter{Interior-point method for quadratic programs in optimal control}
\label{Chapter::Theoretical_Background} % Outline text
\setfigpath{Theory}
In this chapter, the theoretical background regarding the interior-point \gls{QP} solvers is introduced.
The primal--dual \gls{IPM} is introduced first in Section~\ref{sec:ipm}
as the core algorithm for solving \glspl{QP} in this thesis.
Then, the \gls{QP} with \gls{OCP} structure is introduced in Section~\ref{sec:riccati_recursion},
where the Riccati recursion is applied for solving the \gls{QP} with linear complexity.
Finally, the adaptive barrier update and step size control strategies 
are discussed in Section~\ref{sec:update} and Section~\ref{sec:size_control} 
with consideration of the nonconvex case. Algorithm~\ref{alg:ipm} outlines how these parts collaborate.
\begin{algorithm}[htbp]
\caption{Primal--dual \gls{IPM} for \glspl{QP} with \gls{OCP} structure}
\label{alg:ipm}
\begin{algorithmic}[1]
\State \textbf{Input:} \gls{QP} with \gls{OCP} structure, initial iterate, iteration budget $i_{\mathrm{max}}$
\For{$i = 0$ \textbf{to} $i_{\mathrm{max}}-1$}
    \State compute the residuals \Comment{Section~\ref{sec:ipm}}
    \If{all residuals are below their tolerances}
        \State \Return current iterate
    \EndIf
    \State solve with Riccati recursion \Comment{Section~\ref{sec:riccati_recursion}}
    \State update barrier and solve again \Comment{Section~\ref{sec:update}}
    \State compute step sizes and update variables \Comment{Section~\ref{sec:size_control}}
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Primal--dual interior-point method}
\label{sec:ipm}
Consider a \gls{QP} with equality and inequality constraints, of which the formulation can be written as
\begin{mini}[2]
    {y}
    {\frac{1}{2} y^\top H y + g^\top y}
    {\label{eq:qp_problem_unslacked}}
    {}
    \addConstraint{F y}{= b}
    \addConstraint{G y}{\le d},
\end{mini}
where $y$ are the decision variables, $H$ represents the full Hessian, $g$ is the linear term,
and $F$ and $G$ stand for the linear constraint matrices for equality and inequality constraints.
Since the constraints are affine w.r.t.\ $y$, 
the definiteness of the full Hessian matrix $H$ determines whether the \gls{QP} in \eqref{eq:qp_problem_unslacked}
is convex or nonconvex.

By introducing slack variables $t$, we have the equivalent formulation
\begin{mini}[2]
    {y, t}
    {\frac{1}{2} y^\top H y + g^\top y}
    {\label{eq:qp_problem}}
    {}
    \addConstraint{F y}{= b}
    \addConstraint{G y + t}{= d}
    \addConstraint{t}{\ge 0}.
\end{mini}
The Lagrangian function of the \gls{QP} in \eqref{eq:qp_problem} is
\begin{equation}
    \mathcal{L}(y, t, \pi, \lambda)
    = \frac{1}{2} y^\top H y + g^\top y
    + \pi^\top (Fy - b) + \lambda^\top (Gy + t - d),
\end{equation}
where $\pi$ and  $\lambda\ge 0$  are the Lagrange multipliers associated with the equality and inequality constraints, respectively.
Then the \gls{KKT} conditions of the \gls{QP} in \eqref{eq:qp_problem} are given by
% \begin{subequations}\label{eq:KKT_conditions}
% \begin{align}
%     Hy + g + F^\top \pi + G^\top \lambda &= 0, \label{eq:kkt_stat} \\
%     Fy - b &= 0, \label{eq:kkt_eq} \\
%     Gy + t - d &= 0, \label{eq:kkt_ineq} \\
%     \lambda_j\, t_j &= 0, \quad j = 1,\ldots,n_c, \label{eq:kkt_comp} \\
%     \lambda \ge 0, \quad t &\ge 0, \label{eq:kkt_sign}
% \end{align}
% \end{subequations}
\begin{subequations}\label{eq:KKT_conditions}
\begin{align}
    &Hy + g + F^\top \pi + G^\top \lambda = 0, \label{eq:kkt_stat} \\
    &Fy - b = 0, \label{eq:kkt_eq} \\
    &Gy + t - d = 0, \label{eq:kkt_ineq} \\
    &\lambda_j\, t_j = 0, \quad j = 1,\ldots,n_c, \label{eq:kkt_comp} \\
    &\lambda \ge 0, \quad t \ge 0, \label{eq:kkt_sign}
\end{align}
\end{subequations}
where $n_c$ is the number of inequality constraints.
In the primal--dual \gls{IPM}, the complementarity condition \eqref{eq:kkt_comp} is reformulated
as a smooth equation using the barrier parameter $\tau > 0$:
\begin{equation}
    \lambda_j\, t_j = \tau, \quad j = 1,\ldots,n_c, \label{eq:kkt_comp_relaxed}
\end{equation}
and the parameter $\tau$ is shrunk towards zero with a certain strategy during the iterations 
to obtain the solution of the original problem,
and step size control is applied to ensure the satisfaction of the inequalities in \eqref{eq:kkt_sign}.
The corresponding update strategy and step size control are discussed in the following sections.

In order to solve the nonlinear system in \eqref{eq:KKT_conditions},
Newton's method linearizes the system at each iteration $i$
to obtain the step $(\Delta y, \Delta \pi, \Delta \lambda, \Delta t)$ from the linear system, which takes the form of
\begin{equation}
    \begin{bmatrix}
    H & F^\top & G^\top & 0\\
    F & 0 & 0 & 0 \\
    G & 0 & 0 & I \\
    0 & 0 & T_i & \Lambda_i
    \end{bmatrix}
    \begin{bmatrix}
    \Delta y \\
    \Delta \pi \\
    \Delta \lambda \\
    \Delta t
    \end{bmatrix} = -
    \begin{bmatrix}
    r_i^{\mathrm{stat}} \\
    r_i^{\mathrm{eq}} \\
    r_i^{\mathrm{ineq}} \\
    r_i^{\mathrm{comp}}
    \end{bmatrix},
    \label{eq:IPM_KKT_system}
\end{equation}

where the diagonal matrices $T_i=\operatorname{diag}(t_i)$, $\Lambda_i=\operatorname{diag}(\lambda_i)$,
and the perturbed \gls{KKT} residuals follow as

\begin{equation}
    \begin{bmatrix}
        r_i^{\mathrm{stat}} \\ r_i^{\mathrm{eq}} \\
        r_i^{\mathrm{ineq}} \\ r_i^{\mathrm{comp}}
    \end{bmatrix}
    :=
    \begin{bmatrix}
        Hy_i + g + F^\top \pi_i + G^\top \lambda_i \\
        Fy_i - b \\
        Gy_i + t_i - d \\
        T_i \Lambda_i \mathbf{1} - \tau_i \mathbf{1}
    \end{bmatrix}.
    \label{eq:residuals}
\end{equation}

Eliminating $\Delta t$ and $\Delta\lambda$ from the KKT system \eqref{eq:IPM_KKT_system} by a Schur complement 
yields the symmetric saddle point system
\begin{equation}
    \begin{bmatrix}
        \tilde{H}_i & F^\top \\ F & 0
    \end{bmatrix}
    \begin{bmatrix}
        \Delta y \\ \Delta \pi
    \end{bmatrix}
    = -
    \begin{bmatrix}
        \tilde{r}_i^{\mathrm{stat}} \\ r_i^{\mathrm{eq}}
    \end{bmatrix},
    \label{eq:schur_comp_KKT}
\end{equation}
with
$\tilde{H}_i = H + G^\top T_i^{-1}\Lambda_i G \succeq H$ and
$\tilde{r}_i^{\mathrm{stat}} = r_i^{\mathrm{stat}} + G^\top T_i^{-1}(\Lambda_i r_i^{\mathrm{ineq}} - r_i^{\mathrm{comp}})$.
Notably,
the term $G^\top T_i^{-1}\Lambda_i G$ can be regarded 
as the curvature contribution from the inequality constraints.
The weight on each constraint, i.e.\ the corresponding diagonal entry of $T_i^{-1}\Lambda_i$,
increases as the constraint becomes active and tends to zero if it is inactive,
and this barrier term is positive semidefinite, 
since $t$ and $\lambda$ are kept strictly positive throughout the iterations.

In the reduced \gls{KKT} system of \eqref{eq:schur_comp_KKT},
assuming $F$ has full row rank,
the null space method can be applied to further reduce the system.
Let the columns of $Z$ form a basis of the null space of $F$, i.e.\ $FZ = 0$,
and the columns of $Y$ span the range of $F^\top$, 
so that $[\,Y \;\, Z\,]$ is nonsingular and $FY$ is invertible.

The primal step can then be decomposed uniquely as $\Delta y = Y\,\Delta y_y + Z\,\Delta y_z$.
Since $F\Delta y = FY\,\Delta y_y$, only the range-space component enters the equality constraints.
This yields a system exposing the reduced Hessian as
\begin{equation}
    \tilde{M}_i \,\Delta y_z
    = -\big( Z^\top \tilde{r}_i^{\mathrm{stat}}
             + Z^\top \tilde{H}_i\, Y\,\Delta y_y \big),
    \label{eq:schur_comp_KKT_reduced}
\end{equation}
where $Z^\top$ eliminates the equality Jacobian with $Z^\top F^\top = 0$,
feasibility fixes the range-space component $\Delta y_y = -(FY)^{-1} r_i^{\mathrm{eq}}$,
and $\tilde{M}_i = Z^\top (H+G^\top T_i^{-1}\Lambda_i G) Z = Z^\top \tilde{H}_i Z$ is the reduced Hessian including the curvature contribution.

\section{Efficient solution via Riccati recursion}
\label{sec:riccati_recursion}
In the context of optimal control, the decision variables $y$ can be partitioned into state and control variables,
and the \gls{QP} in \eqref{eq:qp_problem_unslacked} takes the particular form
\begin{mini}[2]
    {\substack{x_0, \ldots, x_N \\ u_0, \ldots, u_{N-1}}}
    {\sum_{k=0}^{N-1} \left( \frac{1}{2}
        \begin{bmatrix} x_k \\ u_k \end{bmatrix}^\top
         \underbrace{\begin{bmatrix} Q_k & S_k^\top \\ S_k & R_k \end{bmatrix}}_{H_k}
        \begin{bmatrix} x_k \\ u_k \end{bmatrix}
        + \begin{bmatrix} g_k^x \\ g_k^u \end{bmatrix}^\top
        \begin{bmatrix} x_k \\ u_k \end{bmatrix} \right)}
    {\label{eq:ocp_qp}}
    {}
    \breakObjective{+ \frac{1}{2} x_N^\top Q_N x_N + (g_N^x)^\top x_N}
    \addConstraint{x_0}{= \hat{x}_0}
    \addConstraint{x_{k+1}}{= A_k x_k + B_k u_k + b_k,\quad}{k = 0, \ldots, N-1}
    \addConstraint{G_k^x x_k + G_k^u u_k}{\le d_k,}{k = 0, \ldots, N-1}
    \addConstraint{G_N^x x_N}{\le d_N,}
\end{mini}
where $x_k \in \mathbb{R}^{n_x}$ and $u_k \in \mathbb{R}^{n_u}$ are the state and control at stage $k$, and $N$ is the horizon length.
The $Q_k \in \mathbb{R}^{n_x \times n_x}$, $R_k \in \mathbb{R}^{n_u \times n_u}$, and $S_k \in \mathbb{R}^{n_u \times n_x}$
are the state, control, and state--control coupling blocks of the full Hessian block $H_k$,
$g_k^x \in \mathbb{R}^{n_x}$ and $g_k^u \in \mathbb{R}^{n_u}$ are the corresponding gradients, and $Q_N$, $g_N^x$ are the terminal Hessian and gradient.
The $A_k \in \mathbb{R}^{n_x \times n_x}$ and $B_k \in \mathbb{R}^{n_x \times n_u}$ are the Jacobians of the dynamics with respect to the state and control,
$b_k \in \mathbb{R}^{n_x}$ is the dynamics residual, and $\hat{x}_0 \in \mathbb{R}^{n_x}$ is the initial state.
The $G_k^x \in \mathbb{R}^{n_{k}^c \times n_x}$ and $G_k^u \in \mathbb{R}^{n_{k}^c \times n_u}$ are the Jacobians of the $n_{k}^c$ inequality constraints at stage $k$,
including linear constraints and box constraints on states and controls,
with right-hand side $d_k \in \mathbb{R}^{n_{k}^c}$.

Following the same derivation as in Section \ref{sec:ipm},
we can obtain the \gls{KKT} system of \eqref{eq:ocp_qp},
which exhibits a tridiagonal structure since the cost and inequality constraints are stage-wise 
and the dynamics couple only consecutive stages:
\begin{equation}
    \begin{bNiceMatrix}[margin]
        \ddots & & & & \\
        & 0  & -I \\
        & -I & \tilde{Q}_k & \tilde{S}_k^\top & A_k^\top  \\
        & 0  & \tilde{S}_k &    \tilde{R}_k   & B_k^\top \\
        & 0  &    A_k      &       B_k        &  0 & -I \\
        &    &             &                  & -I & \ddots
    \CodeAfter
        \SubMatrix({3-3}{4-4})
    \end{bNiceMatrix}
    \begin{bmatrix}
        \vdots \\
        \Delta \pi_k \\
        \Delta x_k \\
        \Delta u_k \\
        \Delta \pi_{k+1}\\
        \vdots
    \end{bmatrix} = -
    \begin{bmatrix}
        \vdots \\
        r_{k-1}^{\mathrm{eq}} \\
        \tilde{r}_k^{\mathrm{stat}, x} \\
        \tilde{r}_k^{\mathrm{stat}, u} \\
        r_k^{\mathrm{eq}} \\
        \vdots
    \end{bmatrix},
\end{equation}
where the block matrices $\tilde{Q}_k$, $\tilde{R}_k$, and $\tilde{S}_k$ contain the curvature contributions from inequality constraints as
\begin{align*}
    \tilde{Q}_k &= Q_k + (G_k^x)^\top \Lambda_k T_k^{-1} G_k^x,\\
    \tilde{R}_k &= R_k + (G_k^u)^\top \Lambda_k T_k^{-1} G_k^u,\\
    \tilde{S}_k &= S_k + (G_k^u)^\top \Lambda_k T_k^{-1} G_k^x.
\end{align*}

This tridiagonal structure allows for an efficient solution of the \gls{KKT} system 
using the Riccati recursion~\cite{rao1998application},
which consists of a backward and forward pass.
The backward pass computes the feedback gains $K_k$ and feedforward terms $k_k$,
as well as the cost-to-go matrices $P_k$ and vectors $p_k$,
while the forward pass computes the optimal state and control trajectories.
The algorithmic details of the Riccati recursion are summarized in Algorithm~\ref{alg:riccati_recursion}.
\begin{algorithm}[htbp]
\caption{Riccati recursion for solving a \gls{QP} with OCP structure in \eqref{eq:ocp_qp}}
\label{alg:riccati_recursion}
\begin{algorithmic}[1]
\State\textbf{Initialize:} $P_N \gets \tilde{Q}_N$, $p_N \gets \tilde{r}_N^{\mathrm{stat}, x}$

\Statex
\Statex\textbf{// Backward recursion}
\For{$k = N-1$ \textbf{down to} $0$}
    \State $\tilde{M}_k \gets \tilde{R}_k + B_k^\top P_{k+1} B_k$
    \State $\Gamma_k \gets \tilde{S}_k + B_k^\top P_{k+1} A_k$
    \State $K_k \gets -\tilde{M}_k^{-1} \Gamma_k$
    \State $k_k \gets -\tilde{M}_k^{-1} \left(\tilde{r}_k^{\mathrm{stat}, u} + B_k^\top (P_{k+1} r_k^{\mathrm{eq}} + p_{k+1})\right)$
    \State $P_k \gets \tilde{Q}_k + A_k^\top P_{k+1} A_k - \Gamma_k^\top \tilde{M}_k^{-1} \Gamma_k$
    \State $p_k \gets \tilde{r}_k^{\mathrm{stat}, x} + A_k^\top (P_{k+1} r_k^{\mathrm{eq}} + p_{k+1}) + \Gamma_k^\top k_k$
\EndFor

\Statex
\Statex \textbf{// Forward recursion}
\State $\Delta x_0 \gets \hat{x}_0 - x_0$
\For{$k = 0$ \textbf{to} $N-1$}
    \State $\Delta u_k \gets K_k \Delta x_k + k_k$
    \State $\Delta x_{k+1} \gets A_k \Delta x_k + B_k \Delta u_k + r_k^{\mathrm{eq}}$
    \State $\Delta \pi_{k+1} \gets P_{k+1} \Delta x_{k+1} + p_{k+1}$
    \State $\Delta t_k \gets -(G_k^x \Delta x_k + G_k^u \Delta u_k + r_k^{\mathrm{ineq}})$
    \State $\Delta \lambda_k \gets T_k^{-1} (-r_k^{\mathrm{comp}} - \Lambda_k \Delta t_k)$
\EndFor
\State $\Delta t_N \gets -(G_N^x \Delta x_N + r_N^{\mathrm{ineq}})$
\State $\Delta \lambda_N \gets T_N^{-1} (-r_N^{\mathrm{comp}} - \Lambda_N \Delta t_N)$

\end{algorithmic}
\end{algorithm}

By applying the Riccati recursion, we solve the \gls{KKT} system in \eqref{eq:schur_comp_KKT_reduced} 
without factorizing the reduced Hessian $\tilde{M}_i$ as a whole,
and the computational complexity is reduced from $\mathcal{O}(N^3)$ for 
a dense factorization to $\mathcal{O}(N)$ \cite{rao1998application}, 
thus achieving an efficient solution by exploiting the \gls{OCP} structure.

\section{Adaptive barrier update}
\label{sec:update}
In this section, two closely related adaptive barrier update schemes are discussed for the primal--dual \gls{IPM}.
While monotone updates of the barrier parameter $\tau$,
such as the Fiacco--McCormick scheme \cite{fiacco1990nonlinear}, enjoy strong convergence guarantees,
they may lead to slow convergence and practical limitations \cite{nocedalAdaptiveBarrierUpdate2009a}.
We therefore prefer more aggressive, adaptive updates of $\tau$ for faster convergence,
such as Mehrotra's predictor--corrector and Mehrotra's probing presented next.
Both schemes set $\tau$ adaptively from an affine predictor step,
and the predictor--corrector scheme additionally corrects the search direction.
Section~\ref{sec:barrier_nonconvex} then compares the two schemes for nonconvex \glspl{QP}.

\subsection{Mehrotra's predictor--corrector}
\label{sec:mehrotra}
For convex \glspl{QP}, 
Mehrotra's predictor--corrector scheme \cite{Mehrotra1992implementation}, 
which adaptively updates the barrier parameter $\tau$ and adds a second-order corrector for improving the direction,
is usually considered a good choice for fast convergence.
In this scheme, the barrier parameter $\tau$ is defined as
\begin{align}
    \tau_i = \sigma_i \mu_i, \\
    \mu_i = \frac{t_i^\top \lambda_i}{n_c},
\end{align}
where $\mu_i$ is the duality measure, i.e.\ the average complementarity product,
which measures the violation of the complementarity condition \eqref{eq:kkt_comp},
and $\sigma_i \geq 0$ is the centering parameter that controls the trade-off between centering and aggressive steps.
In each iteration, two solves of \eqref{eq:IPM_KKT_system} are performed
that share the same \gls{LHS} and differ only in the complementarity at the \gls{RHS}, 
and each iteration consists of a predictor (affine scaling) step, a centering update, and a corrector step.

\paragraph{Predictor step.}
The scheme starts with computing a predictor step, also called the affine scaling step.
For this step, the linear system in \eqref{eq:IPM_KKT_system} is solved
with $\sigma_i=0$ in the complementarity residual $r_i^{\mathrm{comp}}$ and no correction considered,
i.e.\ the resulting complementarity residual is written as
\begin{equation}
    \label{eq:predictor_complementarity}
    r_i^{\mathrm{comp}, \mathrm{aff}}
    = T_i\Lambda_i\mathbf{1} - \sigma_i\mu_i\mathbf{1}
    = T_i\Lambda_i\mathbf{1}.
\end{equation}
This yields the affine scaling direction $(\Delta y_{\mathrm{aff}},\Delta \pi_{\mathrm{aff}},\Delta\lambda_{\mathrm{aff}},\Delta t_{\mathrm{aff}})$.
The affine step is used to estimate the centering parameter $\sigma_i$ and form the second-order correction in the following.

\paragraph{Centering update.}
After the predictor step is obtained, the centering parameter $\sigma$ is updated 
based on the estimated duality measure $\mu_{\mathrm{aff}}$ after the affine step is taken,
which is defined as
\begin{align}
    \sigma_i &= (\mu_{\mathrm{aff}}/\mu_i)^3 \label{eq:centering} \\
    \mu_{\mathrm{aff}} &= \frac{(t_i + \alpha_i^{\mathrm{prim}, \mathrm{aff}} \Delta t_{\mathrm{aff}})^\top (\lambda_i + \alpha_i^{\mathrm{dual}, \mathrm{aff}} \Delta \lambda_{\mathrm{aff}})}{n_c},
\end{align}
where the step sizes
$\alpha_i^{\mathrm{prim}, \mathrm{aff}}$ and $\alpha_i^{\mathrm{dual}, \mathrm{aff}}$ are 
the largest step sizes that keep the slack variables $t_i$ and Lagrange multipliers $\Lambda_i$ nonnegative,
i.e.
\begin{align}
    \label{eq:max_step_size}
    \alpha_i^{\mathrm{prim}, \mathrm{aff}} &= \min\left(1, \min_{j:\Delta t_j^{\mathrm{aff}}<0} -\frac{t_{j,i}}{\Delta t_j^{\mathrm{aff}}}\right), \\
    \alpha_i^{\mathrm{dual}, \mathrm{aff}} &= \min\left(1, \min_{j:\Delta \lambda_j^{\mathrm{aff}}<0} -\frac{\lambda_{j,i}}{\Delta \lambda_j^{\mathrm{aff}}}\right).
\end{align}

If $\mu_{\mathrm{aff}} \ll \mu_i$, 
the affine step achieves a substantial reduction in the duality gap,
which drives $\sigma_i$ sharply to $0$, resulting in a near-Newton, aggressive step,
whereas a stalled affine step with $\mu_{\mathrm{aff}} \approx \mu_i$
gives $\sigma_i \to 1$ and thus a recentering step towards the central path.

\paragraph{Corrector step.}
In the corrector step, the linear system in \eqref{eq:IPM_KKT_system} is solved again
where the complementarity residual is modified to 
include the updated centering parameter $\sigma_i$ 
and a second-order correction term using the calculated affine steps as follows:
\begin{equation}
    r_i^{\mathrm{comp},\mathrm{cor}}
    = T_i\Lambda_i\mathbf{1} + \Delta T_{\mathrm{aff}}\Delta\Lambda_{\mathrm{aff}}\mathbf{1} - \sigma_i\mu_i\mathbf{1},
    \label{eq:rcomp_cc}
\end{equation}
where $\Delta T_{\mathrm{aff}} = \operatorname{diag}({\Delta t_{\mathrm{aff}}})$, $\Delta \Lambda_{\mathrm{aff}} = \operatorname{diag}({\Delta \lambda}_{\mathrm{aff}})$,
the term $\Delta T_{\mathrm{aff}}\Delta\Lambda_{\mathrm{aff}}\mathbf{1}$ cancels
the second-order error that the affine step leaves in $t_j\lambda_j$,
and $-\sigma_i\mu_i\mathbf{1}$ supplies the centering.
This second solve returns the final direction $(\Delta y,\Delta\pi,\Delta\lambda,\Delta t)$ directly,
and a step size control follows to ensure the positivity of the slack variables and Lagrange multipliers for the inequality constraints.

\paragraph{Condition on corrector step.}
In practice, the corrector step is usually applied conditionally.
The mechanism is, therefore, called conditional Mehrotra's predictor--corrector scheme \cite{nocedalAdaptiveBarrierUpdate2009a},
where the corrector step is only accepted 
if it does not largely increase the duality measure 
after the corrector step is taken compared with that after the predictor step.
More specifically, the duality measure after the corrector step is estimated as
\begin{equation}
    \mu_{\mathrm{cor}} = \frac{(t_i + \alpha_i^{\mathrm{prim}, \mathrm{cor}} \Delta t)^\top (\lambda_i + \alpha_i^{\mathrm{dual}, \mathrm{cor}} \Delta \lambda)}{n_c},
\end{equation}
where $\alpha_i^{\mathrm{prim}, \mathrm{cor}}$ and $\alpha_i^{\mathrm{dual}, \mathrm{cor}}$ are computed 
as the affine step sizes in \eqref{eq:max_step_size},
but along the corrected direction.
If $\mu_{\mathrm{cor}} \leq 2\mu_{\mathrm{aff}}$, the corrected direction is accepted.
Otherwise, the linear system in \eqref{eq:IPM_KKT_system} is solved again 
without the correction term $\Delta T_{\mathrm{aff}}\Delta\Lambda_{\mathrm{aff}}\mathbf{1}$ in \eqref{eq:rcomp_cc}.

\subsection{Mehrotra's probing}
\label{subsec:mehrotra_probing}
In contrast to Mehrotra's predictor--corrector scheme or its conditional variant,
Mehrotra's probing scheme \cite{nocedalAdaptiveBarrierUpdate2009a} drops the corrector step entirely,
resulting in a more conservative approach for updating the barrier parameter $\tau$.

In Mehrotra's probing,
after one solve of the linear system with \eqref{eq:predictor_complementarity} as the complementarity residual,
another solve gives the so-called centering step, 
and the corresponding complementarity residual at the \gls{RHS} is given by
\begin{equation}
    r_i^{\mathrm{comp}, \mathrm{pro}}
    = T_i\Lambda_i\mathbf{1} - \sigma_i\mu_i\mathbf{1},
\end{equation}
where the centering parameter $\sigma_i$ is updated using \eqref{eq:centering} after the affine step.

\subsection{Comparison of schemes for nonconvex QPs}
\label{sec:barrier_nonconvex}
We argue that Mehrotra's predictor--corrector scheme is not necessarily the best choice in the nonconvex case.
In the corrector step, $\Delta T_{\mathrm{aff}}\Delta\Lambda_{\mathrm{aff}}\mathbf{1}$ is added to the complementarity residual
to include the complementarity error of a full affine step, which assumes the affine direction is an accurate Newton step.
In nonconvex \glspl{QP}, however, the Hessian at the \gls{LHS} usually needs to be modified,
and thus the direction can be adjusted for descent when the constraints are satisfied,
so it is actually an inexact Newton step, violating the assumption of the predictor--corrector scheme.

Probing uses the affine step only through $\sigma = (\mu_{\mathrm{aff}}/\mu)^3$, so an inexact affine step just gives $\sigma \approx 1$, i.e.\ a centering step.
The iterate moves back to the central path instead of trying to decrease the incorrect complementarity error introduced by the inexact Newton step.

\section{Step size control}
\label{sec:size_control}
In this section, we discuss the step size control employed in a primal--dual \gls{IPM}.
The \gls{FTB} rule is first introduced as a common strategy for step size control in \glspl{IPM},
which ensures that the updated slack variables and Lagrange multipliers remain positive.
Then, a two-type line search with backtracking for \glspl{QP} is introduced, which is 
simplified from the funnel line search method introduced in \cite{kiesslingUnifiedFunnelRestoration2025} for the nonconvex case.

\subsection{Fraction-to-the-boundary rule}
\label{subsec:ftb}
The \gls{FTB} rule is a common strategy for step size control in primal--dual \glspl{IPM}.
The maximal step sizes $\alpha_i^{\mathrm{prim},\mathrm{max}}$ and $\alpha_i^{\mathrm{dual},\mathrm{max}}$ are computed
without violating the nonnegativity of the slack variables and Lagrange multipliers, as
\begin{align}
    \alpha_i^{\mathrm{prim},\mathrm{max}} &= \min_{j:\Delta t_j<0} -\frac{t_{j,i}}{\Delta t_j}, \label{eq:ftb_max_prim} \\
    \alpha_i^{\mathrm{dual},\mathrm{max}} &= \min_{j:\Delta \lambda_j<0} -\frac{\lambda_{j,i}}{\Delta \lambda_j}. \label{eq:ftb_max_dual}
\end{align}
Then, the actual step sizes are chosen as a fraction of the maximum step sizes, i.e.,
\begin{align}
    \alpha_i^{\mathrm{prim}} &= \min(1, \eta \alpha_i^{\mathrm{prim},\mathrm{max}}),\\
    \alpha_i^{\mathrm{dual}} &= \min(1, \eta \alpha_i^{\mathrm{dual},\mathrm{max}}),
\end{align}
where $\eta \in (0, 1)$ is a parameter that controls the fraction of the maximum step size to be taken,
typically set to a value close to 1 to ensure sufficient progress while maintaining feasibility.

\subsection{Two-type line search for quadratic programs}
\label{subsec:two_phase_search}
% In nonconvex cases, the effect of line search still can be investigated.
A line search with backtracking can further reduce the step size for a sufficient decrease of the primal infeasibility.
In the nonconvex case, the regularized direction is not the exact Newton step,
and the \gls{FTB} step along a descent direction might still increase the barrier objective in an \gls{IPM}.
We simplify the funnel method proposed in~\cite{kiesslingUnifiedFunnelRestoration2025} 
to a two-type line search tailored for \glspl{QP}, and the two-type line search is executed after the \gls{FTB} part is performed. 

In an \gls{IPM} for \glspl{QP}, the objective $\phi$ used in the line search is the barrier objective
\begin{equation}
    \phi(y, t) = \frac{1}{2} y^\top H y + g^\top y - \tau \sum_{j=1}^{n_c} \ln t_j,
    \label{eq:barrier_objective}
\end{equation}
which is minimized subject to $Fy = b$ and $Gy + t = d$.
With $\lambda_j = \tau / t_j$, the corresponding \gls{KKT} conditions recover the relaxed complementarity in \eqref{eq:kkt_comp_relaxed}.
The line search consists of three parts: a switching condition, the f-type step and the h-type step.

The trial step is classified by the \emph{switching condition}:
it is an \emph{f-type} step if
\begin{equation}
    \label{eq:switching}
    -\alpha_i^{\mathrm{prim}}\rho_i > \kappa_{\mathrm{s}}\,\theta_i^2, \qquad \kappa_{\mathrm{s}} > 0,
\end{equation}
holds, and an \emph{h-type} step otherwise,
where $\theta_i = \lVert (r_i^{\mathrm{eq}}, r_i^{\mathrm{ineq}}) \rVert$ is the constraint violation at the current iterate and
$\rho_i = \nabla \phi_i^\top \Delta z$ the directional derivative of $\phi$ along the primal search direction $\Delta z = (\Delta y, \Delta t)$.
The condition depends only on the current iterate and the trial step size, so it is evaluated before a candidate is formed.

For an h-type step, the predicted objective decrease is small relative to the constraint violation.
The \gls{FTB} step is then accepted without evaluating the objective.
The constraints of the \gls{QP} are affine and the Newton direction solves the corresponding rows of the linear system exactly,
so $\theta(\alpha_i^{\mathrm{prim}}) = (1-\alpha_i^{\mathrm{prim}})\,\theta_i$
and the largest admissible step reduces the constraint violation the most.

For an f-type step, sufficient objective progress is expected.
A candidate iterate $z^{+}(\alpha_i^{\mathrm{prim}})$ is formed in the primal variables $(y,t)$
and accepted if its objective $\phi^{+}$ satisfies the Armijo condition \cite{armijo1966minimization}
\begin{equation}
    \label{eq:armijo}
    \phi^{+} \le \phi_i + \gamma\,\alpha_i^{\mathrm{prim}}\,\rho_i, \qquad \gamma \in (0,1).
\end{equation}
Otherwise, the step size $\alpha_i^{\mathrm{prim}}$ is reduced by $\beta \in (0,1)$ and the search repeats.
If all $n_{\mathrm{max}}$ trials are rejected, the \gls{FTB} step size is restored.
The dual step size $\alpha_i^{\mathrm{dual}}$ is not changed.
The procedure is summarized in Algorithm~\ref{alg:two_phase_line_search}.

\begin{algorithm}[htbp]
\caption{Two-type line search}
\label{alg:two_phase_line_search}
\begin{algorithmic}[1]
\State \textbf{Input:} iterate $z_i$, direction $\Delta z$, \gls{FTB} steps $\alpha_i^{\mathrm{prim}},\alpha_i^{\mathrm{dual}}$, $\phi_i$, $\theta_i$, $\rho_i$
\State \textbf{Parameters:} $\kappa_{\mathrm{s}} > 0$, $\gamma, \beta \in (0,1)$, $n_{\mathrm{max}} \in \mathbb{N}$
\State $\alpha_i^{\mathrm{prim},\mathrm{FTB}} \gets \alpha_i^{\mathrm{prim}}$
\For{$\text{ls} = 1$ \textbf{to} $n_{\mathrm{max}}$}
    \If{$\alpha_i^{\mathrm{prim}}(-\rho_i) \le \kappa_{\mathrm{s}}\,\theta_i^2$} \Comment{h-type: feasibility dominates}
        \State \textbf{break} \Comment{accept \gls{FTB}-scaled step}
    \EndIf
    \State form candidate $z^{+}(\alpha_i^{\mathrm{prim}})$
    \State compute $\phi^{+}$
    \If{$\phi^{+} \le \phi_i + \gamma\,\alpha_i^{\mathrm{prim}}\,\rho_i$} \Comment{f-type: Armijo}
        \State \textbf{break}
    \EndIf
    \If{$\text{ls} = n_{\mathrm{max}}$}
        \State $\alpha_i^{\mathrm{prim}} \gets \alpha_i^{\mathrm{prim},\mathrm{FTB}}$ \Comment{all trials rejected: revert to the \gls{FTB} step}
    \Else
        \State $\alpha_i^{\mathrm{prim}} \gets \beta\,\alpha_i^{\mathrm{prim}}$
    \EndIf
\EndFor
\State \Return $\alpha_i^{\mathrm{prim}},\ \alpha_i^{\mathrm{dual}}$
\end{algorithmic}
\end{algorithm}

Two simplifications distinguish this procedure from the funnel method of \cite{kiesslingUnifiedFunnelRestoration2025}.
First, no funnel bound $\theta_{\mathrm{max}}$ is maintained.
In the funnel method, a trial iterate is admissible only if $\theta(z) \le \theta_{\mathrm{max}}$,
and the bound $\theta_{\mathrm{max}}$ is decreased whenever an h-step is accepted.
Here $\theta(\alpha_i^{\mathrm{prim}}) = (1-\alpha_i^{\mathrm{prim}})\,\theta_i \le \theta_i$ for every $\alpha_i^{\mathrm{prim}} \in (0,1]$,
so no trial step increases the constraint violation, and the funnel itself can then be removed.
Second, since $H$ is generally indefinite, 
$\Delta y^\top H \Delta y$ can have either sign, and the second-order term is unreliable for classifying the step.
Instead of considering the second-order model as in \cite{kiesslingUnifiedFunnelRestoration2025},
the switching condition in \eqref{eq:switching} uses only the first-order term $-\rho_i$, as the Armijo condition in \eqref{eq:armijo} does.
To compensate for this, $\kappa_{\mathrm{s}}$ is set small, so that objective-reducing steps are not classified as h-type steps.
